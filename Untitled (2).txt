def read(path: Path) -> str:
    return path.read_text(encoding="utf-8")


def write(path: Path, text: str) -> None:
    path.write_text(text, encoding="utf-8")


def replace_once(
    path: Path,
    old: str,
    new: str,
    label: str,
) -> None:
    text = read(path)

    if old not in text:
        raise SystemExit(
            f"Could not patch {label} in {path}.\n"
            f"Restore from: {backup_root}"
        )

    write(
        path,
        text.replace(old, new, 1),
    )

def create_light_source_insert(
    parent_object: bpy.types.Object,
    parameters: TileParameters,
) -> bpy.types.Object:
    radius_m = (
        parameters.shape_width_mm
        * 0.36
        * MM_TO_METERS
    )

    housing_height_mm = max(
        2.0,
        parameters.shape_height_mm * 0.65,
    )

    insert_height_mm = max(
        0.35,
        parameters.shape_height_mm * 0.08,
    )

    bpy.ops.mesh.primitive_cylinder_add(
        vertices=parameters.circle_segments,
        radius=radius_m,
        depth=insert_height_mm * MM_TO_METERS,
        location=(
            parent_object.location.x,
            parent_object.location.y,
            parent_object.location.z
            + (
                parameters.tile_height_mm
                + housing_height_mm
                + insert_height_mm
            )
            * MM_TO_METERS,
        ),
    )

    insert = bpy.context.active_object
    insert.name = "PRIM-0011_Emissive_Insert"

    insert.data.materials.append(
        get_or_create_light_material()
    )

    insert["owde_visual_detail"] = True
    insert["owde_detail_type"] = "emissive_light_surface"

    insert.parent = parent_object
    insert.matrix_parent_inverse = (
        parent_object.matrix_world.inverted()
    )

    return insert
