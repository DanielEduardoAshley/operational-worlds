import re
import textwrap

repo = Path(".")
builders_path = repo / "owde_addon" / "core" / "field_builders.py"
fields_path = repo / "owde_addon" / "blender" / "fields.py"

build_path = repo / "scripts" / "build_addon.py"
tests_path = repo / "tests" / "test_field_builders.py"

for path in (builders_path, fields_path):
    if not path.exists():
        raise SystemExit(
            f"Missing required file: {path}\n"
            "Run this from the OWDE repository root."
        )

builders_text = builders_path.read_text(encoding="utf-8")
fields_text = fields_path.read_text(encoding="utf-8")

builders_backup = builders_path.with_suffix(".py.before-field-batch02")
fields_backup = fields_path.with_suffix(".py.before-field-batch02")

builders_backup.write_text(builders_text, encoding="utf-8")
fields_backup.write_text(fields_text, encoding="utf-8")

if "def build_nested_zones_field(" not in builders_text:

    block = textwrap.dedent(
        '''

        def build_nested_zones_field(
            *,
            field_width_mm: float = DEFAULT_FIELD_WIDTH_MM,
            field_height_mm: float = DEFAULT_FIELD_HEIGHT_MM,
            line_width_mm: float = DEFAULT_LINE_WIDTH_MM,
            outer_width_mm: float = 220.0,
            outer_height_mm: float = 310.0,
            inner_width_mm: float = 120.0,
            inner_height_mm: float = 190.0,
        ) -> FieldSpec:
            """FIELD-0005 — Nested Zones

            Two centered rectangular boundaries establish an outer and inner zone.
            """

            half_outer_w = outer_width_mm / 2.0
            half_outer_h = outer_height_mm / 2.0

            half_inner_w = inner_width_mm / 2.0
            half_inner_h = inner_height_mm / 2.0

            lines = (
                # outer zone
                FieldLine(-half_outer_w, -half_outer_h, half_outer_w, -half_outer_h, line_width_mm),
                FieldLine(half_outer_w, -half_outer_h, half_outer_w, half_outer_h, line_width_mm),
                FieldLine(half_outer_w, half_outer_h, -half_outer_w, half_outer_h, line_width_mm),
                FieldLine(-half_outer_w, half_outer_h, -half_outer_w, -half_outer_h, line_width_mm),

                # inner zone
                FieldLine(-half_inner_w, -half_inner_h, half_inner_w, -half_inner_h, line_width_mm),
                FieldLine(half_inner_w, -half_inner_h, half_inner_w, half_inner_h, line_width_mm),
                FieldLine(half_inner_w, half_inner_h, -half_inner_w, half_inner_h, line_width_mm),
                FieldLine(-half_inner_w, half_inner_h, -half_inner_w, -half_inner_h, line_width_mm),
            )

            return FieldSpec(
                field_id="FIELD-0005",
                name="Nested Zones",
                width_mm=field_width_mm,
                height_mm=field_height_mm,
                lines=lines,
                parameters={
                    "line_width_mm": line_width_mm,
                    "outer_width_mm": outer_width_mm,
                    "outer_height_mm": outer_height_mm,
                    "inner_width_mm": inner_width_mm,
                    "inner_height_mm": inner_height_mm,
                },
            )


        def build_offset_zones_field(
            *,
            field_width_mm: float = DEFAULT_FIELD_WIDTH_MM,
            field_height_mm: float = DEFAULT_FIELD_HEIGHT_MM,
            line_width_mm: float = DEFAULT_LINE_WIDTH_MM,
            zone_width_mm: float = 150.0,
            zone_height_mm: float = 230.0,
            offset_x_mm: float = 72.0,
            offset_y_mm: float = 58.0,
        ) -> FieldSpec:
            """FIELD-0006 — Offset Zones

            Two equal rectangular zones displaced diagonally from one another.
            """

            half_w = zone_width_mm / 2.0
            half_h = zone_height_mm / 2.0

            centers = (
                (-offset_x_mm / 2.0, -offset_y_mm / 2.0),
                ( offset_x_mm / 2.0,  offset_y_mm / 2.0),
            )

            lines = []

            for cx, cy in centers:
                left = cx - half_w
                right = cx + half_w
                bottom = cy - half_h
                top = cy + half_h

                lines.extend(
                    (
                        FieldLine(left, bottom, right, bottom, line_width_mm),
                        FieldLine(right, bottom, right, top, line_width_mm),
                        FieldLine(right, top, left, top, line_width_mm),
                        FieldLine(left, top, left, bottom, line_width_mm),
                    )
                )

            return FieldSpec(
                field_id="FIELD-0006",
                name="Offset Zones",
                width_mm=field_width_mm,
                height_mm=field_height_mm,
                lines=tuple(lines),
                parameters={
                    "line_width_mm": line_width_mm,
                    "zone_width_mm": zone_width_mm,
                    "zone_height_mm": zone_height_mm,
                    "offset_x_mm": offset_x_mm,
                    "offset_y_mm": offset_y_mm,
                },
            )


        def build_terminal_target_field(
            *,
            field_width_mm: float = DEFAULT_FIELD_WIDTH_MM,
            field_height_mm: float = DEFAULT_FIELD_HEIGHT_MM,
            line_width_mm: float = DEFAULT_LINE_WIDTH_MM,
            corridor_width_mm: float = 125.0,
            terminal_y_mm: float = 125.0,
            target_width_mm: float = 185.0,
        ) -> FieldSpec:
            """FIELD-0007 — Terminal / Target

            Two parallel boundaries rise from the lower field and terminate
            against a transverse target boundary.
            """

            half_corridor = corridor_width_mm / 2.0
            half_target = target_width_mm / 2.0

            bottom = -field_height_mm / 2.0
            target_y = terminal_y_mm

            lines = (
                FieldLine(
                    -half_corridor,
                    bottom,
                    -half_corridor,
                    target_y,
                    line_width_mm,
                ),
                FieldLine(
                    half_corridor,
                    bottom,
                    half_corridor,
                    target_y,
                    line_width_mm,
                ),
                FieldLine(
                    -half_target,
                    target_y,
                    half_target,
                    target_y,
                    line_width_mm,
                ),
            )

            return FieldSpec(
                field_id="FIELD-0007",
                name="Terminal / Target",
                width_mm=field_width_mm,
                height_mm=field_height_mm,
                lines=lines,
                parameters={
                    "line_width_mm": line_width_mm,
                    "corridor_width_mm": corridor_width_mm,
                    "terminal_y_mm": terminal_y_mm,
                    "target_width_mm": target_width_mm,
                },
            )
        '''
    )

    builders_text += block

import_candidates = [
    "build_divergence_field,",

        inserted_imports = True
        break

    raise SystemExit("Could not parse Divergence enum item.")

new_items = textwrap.dedent(

            (
                "NESTED_ZONES",
                "Nested Zones",
                "An outer zone containing an inner zone",

            (

                "TERMINAL_TARGET",

            ),
    '''

    fields_text[:match.end()]

    + fields_text[match.end():]

'''

# Find the block following Divergence and insert before final else/creation.

        elif settings.architecture == "NESTED_ZONES":
